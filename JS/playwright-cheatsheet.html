Waiting Strategies

waitFor MethodsCore
// Wait for element state
await page.locator('.spinner').waitFor({ state: 'hidden' });
await page.locator('.result').waitFor({ state: 'visible' });
// states: 'attached'|'detached'|'visible'|'hidden'

// Wait for timeout (avoid if possible)
await page.waitForTimeout(1000);

// Wait for function in page context
await page.waitForFunction(() =>
  document.readyState === 'complete'
);

await page.waitForFunction(
  selector => !!document.querySelector(selector),
  '.dynamic-content'
);
Wait for EventsEvents
// Wait for network response
const response = await page.waitForResponse('/api/users');
const response = await page.waitForResponse(
  res => res.url().includes('/api') && res.status() === 200
);

// Wait for network request
const req = await page.waitForRequest('**/submit');

// Wait for DOM event
await page.waitForEvent('console');
await page.waitForEvent('popup');
await page.waitForEvent('download');
await page.waitForEvent('filechooser');

// Combine action + wait (race-condition-safe)
const [download] = await Promise.all([
  page.waitForEvent('download'),
  page.locator('#dl-btn').click(),
]);
Actionability ChecksAuto-wait

// ✓ Element is enabled

// Check visibility without assertion

Network Interception & Mocking

Route — Intercept & MockIntercept
// Abort request
await page.route('**/*.png', route => route.abort());

// Fulfill with mock response
await page.route('**/api/users', route => route.fulfill({
  status: 200,
  contentType: 'application/json',
  body: JSON.stringify([{ id: 1, name: 'Alice' }]),
}));

// Continue with modifications
await page.route('**/api/**', async route => {
  const headers = {
    ...route.request().headers(),
    'X-Custom': 'injected',
  };
  await route.continue({ headers });
});

// Fetch and modify response
await page.route('**/api/data', async route => {
  const response = await route.fetch();
  const json = await response.json();
  json.extra = 'injected';
  await route.fulfill({ response, json });
});

// Remove route
await page.unroute('**/*.png');

JavaScript Evaluation in Browser

page.evaluate()Execute
// Run in browser context, return serializable value
const title = await page.evaluate(() => document.title);

// Pass argument
const len = await page.evaluate(sel =>
  document.querySelectorAll(sel).length, 'li'
);

// evaluateHandle — returns live handle
const body = await page.evaluateHandle(() => document.body);

// Element evaluate
const cls = await page.locator('#el').evaluate(
  el => el.className
);

// Expose Node.js function to browser
await page.exposeFunction('sha256', text =>
  crypto.createHash('sha256').update(text).digest('hex')
);
await page.evaluate(async () => {
  const hash = await sha256('hello');
});
Extract Data from DOMScraping
// textContent
const text = await page.locator('h1').textContent();

// innerText (visible text)
const inner = await page.locator('p').innerText();

// innerHTML
const html = await page.locator('div').innerHTML();

// Attribute value
const href = await page.locator('a').getAttribute('href');

// Input value
const val = await page.locator('input').inputValue();

// Select options
const sel = await page.locator('select').inputValue();

// All link hrefs
const hrefs = await page.evaluate(() =>
  [...document.querySelectorAll('a')].map(a => a.href)
);

Console & Page ErrorsLogging
// Listen to console messages
page.on('console', msg => {
  console.log(`[${msg.type()}]`, msg.text());
  for (const arg of msg.args()) {
    console.log(await arg.jsonValue());
  }
});

// Listen to page errors / exceptions
page.on('pageerror', err => {
  console.error('Page error:', err.message);
});

// Listen to worker errors
page.on('crash', () => console.error('Page crashed!'));
