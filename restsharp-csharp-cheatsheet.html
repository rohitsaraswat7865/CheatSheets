        return await _client.GetAsync<User>(request);
    }

Custom Authenticator (Token Refresh Pattern) Advanced
C# — Custom IAuthenticator with auto-refresh
public class TokenAuthenticator : AuthenticatorBase
{
    private readonly string _tokenUrl;
    private readonly string _clientId;
    private readonly string _clientSecret;

    public TokenAuthenticator(string tokenUrl, string clientId, string secret)
        : base("")
    {
        _tokenUrl     = tokenUrl;
        _clientId     = clientId;
        _clientSecret = secret;
    }

    protected override async ValueTask<string> GetAuthenticationString(
        RestClient client, RestRequest request)
    {
        if (HasTokenExpired())
        {
            var tokenClient   = new RestClient(_tokenUrl);
            var tokenRequest  = new RestRequest()
                .AddParameter("grant_type",    "client_credentials")
                .AddParameter("client_id",     _clientId)
                .AddParameter("client_secret", _clientSecret);

            var tokenResponse = await tokenClient
                .PostAsync<TokenResponse>(tokenRequest);

            Token = tokenResponse.AccessToken;
        }
        return $"Bearer {Token}";
    }
}

C# — Various deserialization approaches

Request/Response Interceptor Interceptors
C#
public class LoggingInterceptor : Interceptors.Interceptor
{
    public override ValueTask BeforeRequest(
        RestRequest request, CancellationToken ct)
    {
        Console.WriteLine(
            $"→ {request.Method} {request.Resource}");
        return default;
    }

    public override ValueTask AfterRequest(
        RestResponse response, CancellationToken ct)
    {
        Console.WriteLine(
            $"← {(int)response.StatusCode} ({response.ContentLength}B)");
        return default;
    }
}

// Register interceptor

{
    BaseUrl = new Uri("https://api.example.com"),
    Interceptors = [new LoggingInterceptor()]
};

Error Handling

Safe patterns for handling API errors, timeouts, and deserialization failures
Safe ExecuteAsync Pattern Error Handling
C#
var response = await client
    .ExecuteAsync<User>(new RestRequest("users/1"));

// 1. Network-level failures
if (response.ResponseStatus == ResponseStatus.TimedOut)
    throw new TimeoutException("Request timed out");

if (response.ResponseStatus != ResponseStatus.Completed)
    throw new HttpRequestException(
        $"Network error: {response.ErrorMessage}",
        response.ErrorException);

// 2. HTTP error status codes
if (!response.IsSuccessStatusCode)
{
    throw response.StatusCode switch
    {
        HttpStatusCode.Unauthorized  => new UnauthorizedException(),
        HttpStatusCode.Forbidden    => new ForbiddenException(),
        HttpStatusCode.NotFound     => new NotFoundException("User not found"),
        HttpStatusCode.TooManyRequests => new RateLimitException(),
        _ => new ApiException($"API error: {(int)response.StatusCode}")
    };
}

// 3. Use the data
User user = response.Data!

Reusable API Client Base Class Architecture
C# — Base client pattern for test frameworks (e.g. Playwright + RestSharp)
public abstract class ApiClientBase
{
    protected readonly RestClient Client;

    protected ApiClientBase(string baseUrl, string token)
    {
        var options = new RestClientOptions(baseUrl)
        {
            Authenticator = new JwtAuthenticator(token),
            Timeout       = TimeSpan.FromSeconds(30)
        };
        Client = new RestClient(options,
            configureSerialization: s => s.UseSystemTextJson(
                new JsonSerializerOptions
                    { PropertyNameCaseInsensitive = true }));
    }

    protected async Task<T> GetAsync<T>(string resource,
        Action<RestRequest>? configure = null)
    {
        var req = new RestRequest(resource);
        configure?.Invoke(req);
        var res = await Client.ExecuteAsync<T>(req);
        EnsureSuccess(res);
        return res.Data!;
    }

    protected async Task<T> PostAsync<T>(string resource, object body)
    {
        var req = new RestRequest(resource).AddJsonBody(body);
        var res = await Client.ExecuteAsync<T>(req, Method.Post);
        EnsureSuccess(res);
        return res.Data!;
    }

    private static void EnsureSuccess(RestResponseBase response)
    {
        if (!response.IsSuccessful)
            throw new ApiException(
                $"{(int)response.StatusCode}: {response.Content}");
    }
}

// Usage: derive a typed client per API area
public class UsersClient : ApiClientBase
{
    public UsersClient(string baseUrl, string token)
        : base(baseUrl, token) { }

    public Task<List<User>> GetAllAsync() =>
        GetAsync<List<User>>("users");

    public Task<User> GetByIdAsync(int id) =>
        GetAsync<User>("users/{id}",
            r => r.AddUrlSegment("id", id));

    public Task<User> CreateAsync(CreateUserDto dto) =>
        PostAsync<User>("users", dto);
}
