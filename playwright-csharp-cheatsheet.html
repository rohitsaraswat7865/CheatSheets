Installation & Setup

getting started
Package Installation
# Install NuGet package
dotnet add package Microsoft.Playwright

# Install browsers
pwsh bin/Debug/net8.0/playwright.ps1 install

# Install specific browser
pwsh bin/Debug/net8.0/playwright.ps1 install chromium
pwsh bin/Debug/net8.0/playwright.ps1 install firefox
pwsh bin/Debug/net8.0/playwright.ps1 install webkit

# Install with dependencies (Linux)
pwsh bin/Debug/net8.0/playwright.ps1 install --with-deps
Test Base Class (NUnit)
using Microsoft.Playwright.NUnit;

[Parallelizable(ParallelScope.Self)]
[TestFixture]
public class MyTests : PageTest
{
    [Test]
    public async Task BasicTest()
    {
        await Page.GotoAsync("https://example.com");
        await Expect(Page).ToHaveTitleAsync("Example");
    }
}

// Other base classes:
// BrowserTest   → gives Browser
// ContextTest   → gives Context
// PlaywrightTest → gives Playwright
Manual Instantiation
using Microsoft.Playwright;

var playwright = await Playwright.CreateAsync();
var browser    = await playwright.Chromium.LaunchAsync(new() {
    Headless = true,
    SlowMo   = 50
});
var context = await browser.NewContextAsync();
var page    = await context.NewPageAsync();

await page.GotoAsync("https://example.com");

await context.CloseAsync();
await browser.CloseAsync();
playwright.Dispose();

page.Locator("button.primary")

// Combining selectors
page.Locator("css=div >> text=Submit")

// Has/has-not pseudo-class
page.Locator("article:has(h2)")

page.Locator(".item:not(.disabled)")

Dialogs, Popups & Downloads

browser events
Dialogs (alert/confirm/prompt)IDialog
// Auto-accept all dialogs
page.Dialog += async (_, dialog) => await dialog.AcceptAsync();

// Auto-dismiss
page.Dialog += async (_, dialog) => await dialog.DismissAsync();

// Inspect before accepting
page.Dialog += async (_, dialog) => {
    Console.WriteLine(dialog.Type);    // alert|confirm|prompt|beforeunload
    Console.WriteLine(dialog.Message);
    Console.WriteLine(dialog.DefaultValue); // for prompt
    if (dialog.Type == "prompt")
        await dialog.AcceptAsync("my input");
    else
        await dialog.AcceptAsync();
};
Popups & New Pagespopups
// Wait for popup opened by action
var popup = await page.RunAndWaitForPopupAsync(async () => {
    await page.GetByText("Open new tab").ClickAsync();
});
await popup.WaitForLoadStateAsync();
Console.WriteLine(popup.Url);

// Wait for any new page in context
var newPage = await context.WaitForPageAsync();

// Context-level page event
context.Page += (_, page) => Console.WriteLine($"New page: {page.Url}");
DownloadsIDownload
// Wait for download triggered by action
var download = await page.RunAndWaitForDownloadAsync(async () => {
    await page.GetByText("Export CSV").ClickAsync();

// Save to disk
await download.SaveAsAsync("/tmp/export.csv");

// Get suggested filename
var name = download.SuggestedFilename;

// Get path (temp file)
var path = await download.PathAsync();

// Get URL
var url = download.Url;

// Read as stream
var stream = await download.CreateReadStreamAsync();

// If error during download
var failure = await download.FailureAsync(); // null if ok

public class LoginPage(IPage page)
{
    private readonly ILocator _email    = page.GetByLabel("Email");
    private readonly ILocator _password = page.GetByLabel("Password");
    private readonly ILocator _submit   = page.GetByRole(AriaRole.Button, new() { Name = "Sign in" });

    public async Task GotoAsync() =>
        await page.GotoAsync("/login");

    public async Task LoginAsync(string email, string password)
    {
        await _email.FillAsync(email);
        await _password.FillAsync(password);
        await _submit.ClickAsync();
    }
}

// In test:
var loginPage = new LoginPage(Page);
await loginPage.GotoAsync();
await loginPage.LoginAsync("user@test.com", "secret");

// Default timeouts (ms)

page.SetDefaultNavigationTimeout(60_000); // navigation
