NUnit Cheatsheet

Quick reference guide for NUnit testing framework

📦 Basic Setup

Installation

dotnet add package NUnit
dotnet add package NUnit3TestAdapter
dotnet add package Microsoft.NET.Test.Sdk

Basic Test Class

using NUnit.Framework;

[TestFixture]
public class CalculatorTests
{
    [Test]
    public void Add_TwoNumbers_ReturnsSum()
    {
        // Arrange
        var calculator = new Calculator();

        // Act
        var result = calculator.Add(2, 3);

        // Assert
        Assert.AreEqual(5, result);
    }
}

🏷️ Test Attributes

Attribute Description Example
[TestFixture] Marks a class as containing tests [TestFixture] public class MyTests { }
[Test] Marks a method as a test [Test] public void MyTest() { }
[SetUp] Runs before each test [SetUp] public void Init() { }
[TearDown] Runs after each test [TearDown] public void Cleanup() { }
[OneTimeSetUp] Runs once before all tests in fixture [OneTimeSetUp] public void InitAll() { }
[OneTimeTearDown] Runs once after all tests in fixture [OneTimeTearDown] public void CleanupAll() { }
[Ignore("reason")] Skips a test [Ignore("Not implemented")] [Test]
[Category("name")] Groups tests by category [Category("Integration")] [Test]
[TestCase] Provides inline test data [TestCase(2, 3, 5)] public void Add(int a, int b, int expected)

✓ Common Assertions

Equality

Assert.AreEqual(expected, actual);
Assert.AreNotEqual(expected, actual);
Assert.AreSame(expected, actual);
Assert.AreNotSame(expected, actual);

Boolean

Assert.IsTrue(condition);
Assert.IsFalse(condition);
Assert.That(condition, Is.True);
Assert.That(condition, Is.False);

Null/NotNull

Assert.IsNull(object);
Assert.IsNotNull(object);
Assert.That(object, Is.Null);
Assert.That(object, Is.Not.Null);

Exceptions

Assert.Throws<Exception>(() => method());
Assert.DoesNotThrow(() => method());
var ex = Assert.Throws<ArgumentException>(
    () => method());
Assert.That(ex.Message, Contains.Substring("text"));

Collections

Assert.That(list, Is.Empty);
Assert.That(list, Has.Count.EqualTo(5));
Assert.That(list, Contains.Item(item));
Assert.That(list, Has.Member(item));
Assert.That(list, Is.Ordered);

Strings

Assert.That(str, Is.EqualTo("text"));
Assert.That(str, Does.Contain("sub"));
Assert.That(str, Does.StartWith("pre"));
Assert.That(str, Does.EndWith("suf"));
Assert.That(str, Is.Empty);

🎯 Constraint Model (Assert.That)

Comparison Constraints

Assert.That(actual, Is.EqualTo(expected));
Assert.That(actual, Is.Not.EqualTo(expected));
Assert.That(actual, Is.GreaterThan(5));
Assert.That(actual, Is.LessThan(10));
Assert.That(actual, Is.GreaterThanOrEqualTo(5));
Assert.That(actual, Is.LessThanOrEqualTo(10));
Assert.That(actual, Is.InRange(1, 10));

String Constraints

Assert.That(str, Does.Contain("substring"));
Assert.That(str, Does.Not.Contain("substring"));
Assert.That(str, Does.StartWith("prefix"));
Assert.That(str, Does.EndWith("suffix"));
Assert.That(str, Does.Match(@"\d+"));
Assert.That(str, Is.EqualTo("text").IgnoreCase);

Collection Constraints

Assert.That(collection, Is.Empty);
Assert.That(collection, Is.Not.Empty);
Assert.That(collection, Has.Count.EqualTo(5));
Assert.That(collection, Contains.Item(item));
Assert.That(collection, Has.Member(item));
Assert.That(collection, Has.All.GreaterThan(0));
Assert.That(collection, Has.Some.LessThan(10));
Assert.That(collection, Is.Ordered);
Assert.That(collection, Is.Ordered.Descending);
Assert.That(collection, Is.Unique);

Type Constraints

Assert.That(obj, Is.TypeOf<MyClass>());
Assert.That(obj, Is.InstanceOf<IMyInterface>());
Assert.That(obj, Is.AssignableFrom<MyClass>());
Assert.That(obj, Is.AssignableTo<IMyInterface>());

🔢 Parameterized Tests

TestCase

[TestCase(1, 2, 3)]
[TestCase(10, 20, 30)]
[TestCase(-5, 5, 0)]
public void Add_VariousInputs_ReturnsCorrectSum(int a, int b, int expected)
{
    var result = calculator.Add(a, b);
    Assert.AreEqual(expected, result);
}

TestCaseSource

[TestCaseSource(nameof(AddTestCases))]
public void Add_FromSource(int a, int b, int expected)
{
    Assert.AreEqual(expected, calculator.Add(a, b));
}

private static IEnumerable<TestCaseData> AddTestCases()
{
    yield return new TestCaseData(1, 2, 3).SetName("PositiveNumbers");
    yield return new TestCaseData(-1, -2, -3).SetName("NegativeNumbers");
    yield return new TestCaseData(0, 0, 0).SetName("Zeros");
}

Values & ValueSource

[Test]
public void Test_WithValues([Values(1, 2, 3)] int x, [Values(10, 20)] int y)
{
    // This creates 6 test cases (3 * 2)
    Assert.That(x + y, Is.GreaterThan(0));
}

[Test]
public void Test_WithValueSource([ValueSource(nameof(GetValues))] int value)
{
    Assert.That(value, Is.Positive);
}

private static int[] GetValues() => new[] { 1, 2, 3, 4, 5 };

Combinatorial

[Test, Combinatorial]
public void Test_Combinatorial(
    [Values(1, 2)] int x,
    [Values("A", "B", "C")] string y)
{
    // Creates 6 tests (2 * 3)
}

⚡ Async Tests

[Test]
public async Task AsyncMethod_ReturnsExpectedValue()
{
    // Arrange
    var service = new MyService();

    // Act
    var result = await service.GetDataAsync();

    // Assert
    Assert.IsNotNull(result);
}

[Test]
public void AsyncMethod_ThrowsException()
{
    var service = new MyService();
    Assert.ThrowsAsync<InvalidOperationException>(
        async () => await service.FailingMethodAsync());
}

🔄 Test Lifecycle

[TestFixture]
public class TestLifecycleExample
{
    [OneTimeSetUp]
    public void OneTimeSetUp()
    {
        // Runs once before all tests in this fixture
        Console.WriteLine("OneTimeSetUp");
    }

    [SetUp]
    public void SetUp()
    {
        // Runs before each test
        Console.WriteLine("SetUp");
    }

    [Test]
    public void Test1()
    {
        Console.WriteLine("Test1");
    }

    [Test]
    public void Test2()
    {
        Console.WriteLine("Test2");
    }

    [TearDown]
    public void TearDown()
    {
        // Runs after each test
        Console.WriteLine("TearDown");
    }

    [OneTimeTearDown]
    public void OneTimeTearDown()
    {
        // Runs once after all tests in this fixture
        Console.WriteLine("OneTimeTearDown");
    }
}
Execution Order: OneTimeSetUp → (SetUp → Test → TearDown) × n → OneTimeTearDown

⚡ Parallel Test Execution

Assembly Level Parallelization

// In AssemblyInfo.cs or any test file
using NUnit.Framework;

// Enable parallel execution at assembly level
[assembly: Parallelizable(ParallelScope.Fixtures)]

// Set maximum number of workers
[assembly: LevelOfParallelism(4)]

Parallel Scopes

Scope Description
ParallelScope.Self Only this test/fixture runs in parallel with others
ParallelScope.Children Child tests run in parallel with each other
ParallelScope.Fixtures Test fixtures run in parallel (default recommendation)
ParallelScope.All Everything runs in parallel
ParallelScope.None No parallelization

Fixture Level Control

// Enable parallel execution for this fixture
[TestFixture]
[Parallelizable(ParallelScope.Children)]
public class ParallelTests
{
    [Test] public void Test1() { } // Runs in parallel
    [Test] public void Test2() { } // Runs in parallel
}

// Disable parallel execution for this fixture
[TestFixture]
[NonParallelizable]
public class SequentialTests
{
    [Test] public void Test1() { } // Runs sequentially
    [Test] public void Test2() { } // Runs sequentially
}

Test Level Control

[TestFixture]
public class MixedParallelizationTests
{
    [Test]
    [Parallelizable]
    public void ParallelTest1() { }

    [Test]
    [Parallelizable]
    public void ParallelTest2() { }

    [Test]
    [NonParallelizable]
    public void SequentialTest() { }
}

Configuration Examples

// Example 1: Run all fixtures in parallel, tests within fixtures sequential
[assembly: Parallelizable(ParallelScope.Fixtures)]
[assembly: LevelOfParallelism(8)]

// Example 2: Run everything in parallel
[assembly: Parallelizable(ParallelScope.All)]
[assembly: LevelOfParallelism(4)]

// Example 3: Specific fixture with parallel children
[TestFixture]
[Parallelizable(ParallelScope.Children)]
public class DatabaseTests
{
    [Test] public void Test1() { }
    [Test] public void Test2() { }
}
Important: When running tests in parallel, ensure they don't share mutable state. Use thread-safe code and avoid static fields. Tests with database operations or file I/O may need to be marked [NonParallelizable] or use unique resources per test.

Command Line

// Run with specific number of workers
dotnet test -- NUnit.NumberOfTestWorkers=4

// Disable parallel execution from command line
dotnet test -- NUnit.NumberOfTestWorkers=0

▶️ Running Tests

Command Line

// Run all tests
dotnet test

// Run tests with filter
dotnet test --filter "Category=Integration"
dotnet test --filter "FullyQualifiedName~Calculator"

// Run tests with verbosity
dotnet test --logger "console;verbosity=detailed"

// Run tests and collect coverage
dotnet test --collect:"XPlat Code Coverage"

💡 Best Practices

Naming Conventions

  • MethodName_Scenario_ExpectedResult
  • Given_When_Then
  • Should_ExpectedBehavior_When_StateUnderTest

Test Structure

  • Arrange: Set up test data
  • Act: Execute the method
  • Assert: Verify the result

General Tips

  • One assertion per test (when possible)
  • Tests should be independent
  • Use descriptive test names
  • Keep tests fast
  • Don't test private methods directly

Code Organization

  • Group related tests in test fixtures
  • Use categories for test organization
  • Extract common setup to [SetUp]
  • Use TestCaseSource for complex data
