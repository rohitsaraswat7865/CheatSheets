Project Setup

Create a new .NET test project and add the required packages. UriBuilder is built into the BCL — no extra package needed.

  • 1

    Create project: dotnet new nunit -n PlaywrightApiTests

  • 2

    Add NuGet packages below.

  • 3

UriBuilder is in the System namespace — no NuGet needed. Newtonsoft.Json handles all serialization; System.Text.Json is not used in this tutorial.

02

Creating an API Context

Create IAPIRequestContext once in [SetUp]. Declare shared JsonSerializerSettings here too so every test reuses the same Newtonsoft config.

C# — Base test class
using Microsoft.Playwright;
using Microsoft.Playwright.NUnit;
using Newtonsoft.Json;
using Newtonsoft.Json.Serialization;
using NUnit.Framework;

[TestFixture]
public class ApiTestBase : PlaywrightTest
{
    protected IAPIRequestContext Request;

    // Shared Newtonsoft settings — declare once, reuse everywhere
    protected static readonly JsonSerializerSettings JsonSettings =
        new JsonSerializerSettings
        {
            ContractResolver     = new CamelCasePropertyNamesContractResolver(),
            NullValueHandling    = NullValueHandling.Ignore,
            DateFormatHandling   = DateFormatHandling.IsoDateFormat
        };

    [SetUp]
    public async Task SetUpContext()
    {
        Request = await Playwright.APIRequest.NewContextAsync(new()
        {
            BaseURL = "https://jsonplaceholder.typicode.com",
            ExtraHTTPHeaders = new Dictionary<string, string>
            {
                { "Content-Type", "application/json" },
                { "Accept",       "application/json" }
            },
            IgnoreHTTPSErrors = true
        });
    }

    [TearDown]
    public async Task TearDownContext()
        => await Request.DisposeAsync();
}

⚙️ BaseURL

Set once on the context. UriBuilder-built URLs can still use the full absolute URL — Playwright resolves both.

📦 JsonSettings

Static field shared across all tests — CamelCase resolver, null-ignore, ISO dates.

🔒 IgnoreHTTPSErrors

Bypass SSL errors on staging or self-signed cert environments.

⏱ Timeout

Default 30 000 ms. Override with Timeout = 60000 on the context.

03

GET Requests

Construct the URL with UriBuilder, call GetAsync(), and deserialize the body with JsonConvert.DeserializeObject.

C# — Model
public class Post
{
    [JsonProperty("id")]     public int    Id     { get; set; }
    [JsonProperty("title")]  public string Title  { get; set; }
    [JsonProperty("body")]   public string Body   { get; set; }
    [JsonProperty("userId")] public int    UserId { get; set; }
}
C# — GET single resource
[Test]
public async Task Get_SinglePost_ReturnsCorrectData()
{
    // UriBuilder — safe scheme + host + path composition
    var uri = new UriBuilder
    {
        Scheme = "https",
        Host   = "jsonplaceholder.typicode.com",
        Path   = "/posts/1"
    };

    var response = await Request.GetAsync(uri.ToString());

    Assert.That(response.Status, Is.EqualTo(200));
    Assert.That(response.Ok,     Is.True);

    // Deserialize with Newtonsoft
    string body = await response.TextAsync();
    Post post   = JsonConvert.DeserializeObject<Post>(body, JsonSettings);

    Assert.Multiple(() =>
    {
        Assert.That(post.Id,    Is.EqualTo(1));
        Assert.That(post.Title, Is.Not.Empty());
    });
}
C# — GET collection with query string
[Test]
public async Task Get_PostsByUser_ReturnsList()
{
    // UriBuilder.Query appends ?userId=1 without manual concatenation
    var uri = new UriBuilder
    {
        Scheme = "https",
        Host   = "jsonplaceholder.typicode.com",
        Path   = "/posts",
        Query  = "userId=1"
    };

    var response = await Request.GetAsync(uri.ToString());
    Assert.That(response.Status, Is.EqualTo(200));

    string body      = await response.TextAsync();
    List<Post> posts = JsonConvert.DeserializeObject<List<Post>>(body, JsonSettings);

    Assert.That(posts.Count,               Is.GreaterThan(0));
    Assert.That(posts.All(p => p.UserId == 1), Is.True());
}
✅

Always check response.Ok before calling TextAsync() — avoids misleading null-reference errors when the API returns an error body.

04

POST Requests

Serialize the payload with JsonConvert.SerializeObject and pass it as the Data string. Build the endpoint URL with UriBuilder.

C# — POST create resource
[Test]
public async Task Post_CreatePost_Returns201()
{
    // UriBuilder — endpoint URL
    var uri = new UriBuilder
    {
        Scheme = "https",
        Host   = "jsonplaceholder.typicode.com",
        Path   = "/posts"
    };

    // Newtonsoft — serialize request body
    var newPost = new Post
    {
        Title  = "My New Post",
        Body   = "This is the body content",
        UserId = 1
    };
    string requestJson = JsonConvert.SerializeObject(newPost, JsonSettings);

    var response = await Request.PostAsync(uri.ToString(), new APIRequestContextOptions
    {
        Headers = new Dictionary<string, string>
        {
            { "Content-Type", "application/json" }
        },
        Data = requestJson   // use Data, not DataObject, with Newtonsoft
    });

    Assert.That(response.Status, Is.EqualTo(201));

    // Newtonsoft — deserialize response
    string body  = await response.TextAsync();
    Post created = JsonConvert.DeserializeObject<Post>(body, JsonSettings);

    Assert.That(created.Title, Is.EqualTo("My New Post"));
    Assert.That(created.Id,    Is.GreaterThan(0));
}
05

PUT Requests

PUT replaces the entire resource. Inject the resource ID into the path via UriBuilder, and serialize the full object with Newtonsoft.

C# — PUT update resource
[Test]
public async Task Put_UpdatePost_ReturnsUpdatedData()
{
    int postId = 1;

    // UriBuilder — inject ID cleanly into path via string interpolation
    var uri = new UriBuilder
    {
        Scheme = "https",
        Host   = "jsonplaceholder.typicode.com",
        Path   = $"/posts/{postId}"
    };

    // Newtonsoft — serialize full updated object
    var updated = new Post
    {
        Id     = postId,
        Title  = "Updated Title",
        Body   = "Updated body content",
        UserId = 1
    };
    string requestJson = JsonConvert.SerializeObject(updated, JsonSettings);

    var response = await Request.PutAsync(uri.ToString(), new APIRequestContextOptions
    {
        Headers = new Dictionary<string, string>
        {
            { "Content-Type", "application/json" }
        },
        Data = requestJson
    });

    Assert.That(response.Status, Is.EqualTo(200));

    // Newtonsoft — deserialize response
    string body = await response.TextAsync();
    Post result = JsonConvert.DeserializeObject<Post>(body, JsonSettings);

    Assert.Multiple(() =>
    {
        Assert.That(result.Id,    Is.EqualTo(postId));
        Assert.That(result.Title, Is.EqualTo("Updated Title"));
    });
}
⚠️

Never use DataObject alongside Newtonsoft — Playwright serializes DataObject internally with System.Text.Json. Always use Data with a Newtonsoft-serialized string.

Authentication

Three common patterns — set headers on the context so they apply to every request.

C# — Bearer Token + UriBuilder
Request = await Playwright.APIRequest.NewContextAsync(new()
{
    BaseURL = "https://api.example.com",
    ExtraHTTPHeaders = new Dictionary<string, string>
    {
        { "Authorization", $"Bearer {token}" },
        { "Content-Type",  "application/json" }
    }
});

// Compose the URL — auth header is sent automatically
var uri = new UriBuilder
{
    Scheme = "https",
    Host   = "api.example.com",
    Path   = "/api/profile"
};
var response = await Request.GetAsync(uri.ToString());
C# — Basic Auth
string credentials = Convert.ToBase64String(
    Encoding.UTF8.GetBytes("username:password")
);

ExtraHTTPHeaders = new Dictionary<string, string>
{
    { "Authorization", $"Basic {credentials}" }
};
C# — API Key
ExtraHTTPHeaders = new Dictionary<string, string>
{
    { "x-api-key", "your-api-key-here" }
};

Using Playwright Session

Load session.json into the API context for automatic cookie injection. Combine with UriBuilder for URL construction and Newtonsoft for parsing.

C# — session.json + UriBuilder + Newtonsoft
Request = await Playwright.APIRequest.NewContextAsync(new()
{
    BaseURL          = "https://app.example.com",
    StorageStatePath = "session.json"  // cookies injected automatically
});

// UriBuilder — construct the authenticated endpoint
var uri = new UriBuilder
{
    Scheme = "https",
    Host   = "app.example.com",
    Path   = "/api/me"
};

var response = await Request.GetAsync(uri.ToString());
Assert.That(response.Status, Is.EqualTo(200));

// Newtonsoft — deserialize response
string body    = await response.TextAsync();
UserProfile me = JsonConvert.DeserializeObject<UserProfile>(body, JsonSettings);
Assert.That(me.Email, Is.Not.Empty());
C# — Save a session from browser login
// After logging in via browser automation:
await context.StorageStateAsync(new()
{
    Path = "session.json"

UriBuilder — URL Construction

UriBuilder is in the System namespace (built into .NET). It composes URLs property-by-property — no string concatenation, no manual encoding, no risk of malformed URLs.

C# — All UriBuilder properties
var uri = new UriBuilder
{
    Scheme = "https",                      // https://
    Host   = "api.example.com",            // api.example.com
    Port   = 8443,                           // :8443  (omit for default 443)
    Path   = "/v2/users/42/posts",          // /v2/users/42/posts
    Query  = "status=active&page=1"        // ?status=active&page=1
};

Console.WriteLine(uri.ToString());
// → https://api.example.com:8443/v2/users/42/posts?status=active&page=1
C# — Dynamic path with resource ID
int userId = 42;
int postId  = 7;

var uri = new UriBuilder
{
    Scheme = "https",
    Host   = "api.example.com",
    Path   = $"/users/{userId}/posts/{postId}"
};
// → https://api.example.com/users/42/posts/7

var response = await Request.GetAsync(uri.ToString());
string body  = await response.TextAsync();
Post   post  = JsonConvert.DeserializeObject<Post>(body, JsonSettings);
C# — Multiple query params via NameValueCollection
using System.Web;   // HttpUtility.ParseQueryString

var qs = HttpUtility.ParseQueryString(string.Empty);
qs["status"] = "active";
qs["page"]   = "2";
qs["limit"]  = "20";

var uri = new UriBuilder
{
    Scheme = "https",
    Host   = "api.example.com",
    Path   = "/users",
    Query  = qs.ToString()  // special chars are encoded automatically
};
// → https://api.example.com/users?status=active&page=2&limit=20

var response = await Request.GetAsync(uri.ToString());
string body      = await response.TextAsync();
List<User> users = JsonConvert.DeserializeObject<List<User>>(body, JsonSettings);
C# — Reusable helper: BuildUrl()
/// Compose a full URL from host, path, and optional query params.
private static string BuildUrl(string host, string path,
    Dictionary<string, string> queryParams = null)
{
    var uri = new UriBuilder { Scheme = "https", Host = host, Path = path };

    if (queryParams != null)
    {
        var qs = HttpUtility.ParseQueryString(string.Empty);
        foreach (var kv in queryParams) qs[kv.Key] = kv.Value;
        uri.Query = qs.ToString();
    }

    return uri.ToString();
}

// Usage in tests:
string url = BuildUrl("api.example.com", "/users",
    new() { { "role", "admin" }, { "active", "true" } });

var response = await Request.GetAsync(url);
string body       = await response.TextAsync();
List<User> admins = JsonConvert.DeserializeObject<List<User>>(body, JsonSettings);

Serialization with Newtonsoft.Json

All serialize/deserialize operations use Newtonsoft.Json. Configure JsonSerializerSettings once and pass it to every call.

C# — JsonSerializerSettings
private static readonly JsonSerializerSettings JsonSettings = new()
{
    ContractResolver     = new CamelCasePropertyNamesContractResolver(), // camelCase output
    NullValueHandling    = NullValueHandling.Ignore,                        // skip null fields
    DefaultValueHandling = DefaultValueHandling.Ignore,                     // skip 0/false/empty
    DateFormatHandling   = DateFormatHandling.IsoDateFormat,                // ISO 8601 dates
    Formatting           = Formatting.None                                  // compact (no indents)
};
C# — Serialize & Deserialize patterns
// Object → JSON string  (use as request body via Data =)
string json = JsonConvert.SerializeObject(myObject, JsonSettings);

// JSON string → single object  (from response.TextAsync())
Post post = JsonConvert.DeserializeObject<Post>(body, JsonSettings);

// JSON string → collection
List<Post> posts = JsonConvert.DeserializeObject<List<Post>>(body, JsonSettings);
C# — Model annotations
public class User
{
    [JsonProperty("id")]
    public int Id { get; set; }

    [JsonProperty("first_name")]       // snake_case API field → PascalCase C# property
    public string FirstName { get; set; }

    [JsonProperty("email")]
    public string Email { get; set; }

    [JsonProperty("created_at")]
    public DateTime CreatedAt { get; set; }

    [JsonIgnore]                          // never serialized or deserialized
    public string InternalNote { get; set; }
}
C# — Full PUT round-trip (UriBuilder + Newtonsoft)
[Test]
public async Task Put_FullRoundTrip_UriBuilderAndNewtonsoft()
{
    int userId = 5;

    // 1 — Build URL with UriBuilder
    var uri = new UriBuilder
    {
        Scheme = "https",
        Host   = "jsonplaceholder.typicode.com",
        Path   = $"/users/{userId}"
    };

    // 2 — Serialize request body with Newtonsoft
    var payload = new User
    {
        Id        = userId,
        FirstName = "Alice",
        Email     = "alice@example.com"
    };
    string requestJson = JsonConvert.SerializeObject(payload, JsonSettings);

    // 3 — Send PUT
    var response = await Request.PutAsync(uri.ToString(), new APIRequestContextOptions
    {
        Headers = new Dictionary<string, string>
        {
            { "Content-Type", "application/json" }
        },
        Data = requestJson
    });

    Assert.That(response.Ok, Is.True);

    // 4 — Deserialize response with Newtonsoft
    string body  = await response.TextAsync();
    User result  = JsonConvert.DeserializeObject<User>(body, JsonSettings);

    Assert.That(result.Email, Is.EqualTo("alice@example.com"));
}

[JsonIgnore] Excludes property from both serialize and deserialize
[JsonRequired] Throws if the field is missing during deserialization
[JsonConverter(...)] Custom converter for dates, enums, polymorphism
