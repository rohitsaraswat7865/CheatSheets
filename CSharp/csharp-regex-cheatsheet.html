Regex Class Methods
IsMatch · Match · MatchesMethods
// ── IsMatch ──────────────────────────────────
bool found = Regex.IsMatch("Hello World", @"\bWorld\b");
// true

// ── Match (first match) ──────────────────────
Match m = Regex.Match("Price: £299", @"£(\d+)");
if (m.Success)
    Console.WriteLine(m.Groups[1].Value); // 299

// ── Matches (all matches) ────────────────────
var all = Regex.Matches("1 fish 2 fish", @"\d");
foreach (Match hit in all)
    Console.WriteLine(hit.Value);  // "1", "2"

// ── Compiled instance (reuse for performance) ─
private static readonly Regex _re =
    new(@"\d+", RegexOptions.Compiled);
Named Captures & GroupsGroups
var pattern = @"(?<year>\d{4})-(?<month>\d{2})-(?<day>\d{2})";
Match m = Regex.Match("Date: 2024-03-15", pattern);

if (m.Success)
{
    string year  = m.Groups["year"].Value;   // "2024"
    string month = m.Groups["month"].Value;  // "03"
    string day   = m.Groups["day"].Value;    // "15"
}

// Iterate all groups
foreach (Group g in m.Groups)
    Console.WriteLine($"{g.Name}: {g.Value}");

// Capture collection (multiple hits of one group)
var digits = Regex.Match("a1b2c3", @"(\d)+");
foreach (Capture c in digits.Groups[1].Captures)
    Console.Write(c.Value);  // 1 2 3
Replace & SubstitutionReplace
// ── Simple replace ───────────────────────────
string r = Regex.Replace("foo bar", @"\s+", "-");
// "foo-bar"

// ── Substitution tokens ──────────────────────
// $0  entire match     $1 $2   group by index
// ${name}  named group  $`  before match
// $'  after match       $+  last captured group

Regex.Replace("2024-03-15", @"(\d{4})-(\d{2})-(\d{2})",
             "$3/$2/$1");  // "15/03/2024"

// ── MatchEvaluator (lambda) ──────────────────
string result = Regex.Replace("hello world", @"\b\w",
    m => m.Value.ToUpper());
// "Hello World"
SplitSplit
// ── Basic split ──────────────────────────────
string[] parts = Regex.Split("a1b22c333d", @"\d+");
// ["a", "b", "c", "d"]

// ── Keep delimiters (use capturing group) ────
string[] kept = Regex.Split("one, two; three", @"([,;]\s*)");
// ["one", ", ", "two", "; ", "three"]

// ── Limit count ──────────────────────────────
var re = new Regex(@"\s+");
string[] two = re.Split("a b c d", 2);
// ["a", "b c d"]
Modern C# — Source Generators
GeneratedRegex (.NET 7+).NET 7+
// Compile regex at build time — zero startup cost
using System.Text.RegularExpressions;

public partial class Parser
{
    [GeneratedRegex(@"\b\d{4}\b", RegexOptions.IgnoreCase)]
    private static partial Regex FourDigitRegex();

    public void Run(string input)
    {
        var m = FourDigitRegex().Match(input);
        Console.WriteLine(m.Value);
    }
}

// Benefits:  no reflection · AOT-safe · faster startup
Span & Performance TipsPerf
// ── Avoid new Regex() on every call ─────────
private static readonly Regex _email =
    new(@"^[\w.+-]+@[\w-]+\.[a-z]{2,}$",
        RegexOptions.Compiled | RegexOptions.IgnoreCase);

// ── ValueMatchEnumerator (no heap alloc) ────
foreach (var m in FourDigitRegex().EnumerateMatches(input.AsSpan()))
    Console.WriteLine(input.Substring(m.Index, m.Length));

// ── Timeout to prevent ReDoS ─────────────── 
var safe = new Regex(@"(a+)+",
    RegexOptions.None,
    matchTimeout: TimeSpan.FromMilliseconds(100));
